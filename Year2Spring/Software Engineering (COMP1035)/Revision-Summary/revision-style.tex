\definecolor{accent}{HTML}{1E8A8A}    % teal       - H2 / tags
\definecolor{boldcol}{HTML}{14406B}   % bold terms

% ---- colored, well-spaced headings ----
\usepackage{titlesec}
\titleformat{\section}{\Large\bfseries\color{primary}}{\thesection.}{0.5em}{}[{\color{rulecol}\titlerule[1pt]}]
\titleformat{\subsection}{\large\bfseries\color{accent}}{}{0em}{}
\titleformat{\subsubsection}{\normalsize\bfseries\color{primary}}{}{0em}{}
\titlespacing*{\section}{0pt}{18pt}{9pt}
\titlespacing*{\subsection}{0pt}{12pt}{5pt}
\titlespacing*{\subsubsection}{0pt}{8pt}{3pt}

% ---- airy spacing ----
\usepackage{setspace}
\setstretch{1.6}
\setlength{\parskip}{12pt}
\setlength{\parindent}{0pt}
\setlength{\emergencystretch}{3em}

% setspace stretch breaks longtable page splits -> reset inside tables
\usepackage{etoolbox}
\AtBeginEnvironment{longtable}{\setstretch{1.0}\small}

% keep code blocks compact so they never overflow the right margin
\AtBeginEnvironment{verbatim}{\setstretch{1.0}\small}
\AtBeginEnvironment{Highlighting}{\setstretch{1.0}\small}

% ---- colour bold key terms ----
\let\oldtextbf\textbf

% ---- lecture tag badge:  \lec{Lec07} ----

% (incl. the [Lec] badges) never overflow the right margin
